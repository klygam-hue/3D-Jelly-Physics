cmake_minimum_required(VERSION 3.25)
project(3D_Jelly_Physics_Mobile VERSION 1.9.3 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(JELY_CODE "${CMAKE_CURRENT_SOURCE_DIR}/../Code")
option(JELY_TOUCH_TESTS_ONLY "Build portable touch regressions without a mobile SDK" OFF)
if(JELY_TOUCH_TESTS_ONLY)
  enable_testing()
  add_executable(mobile_touch_tests TouchTests.cpp)
  add_test(NAME mobile_touch COMMAND mobile_touch_tests)
  return()
endif()
if(NOT ANDROID AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  message(FATAL_ERROR "Use the Android NDK toolchain or CMAKE_SYSTEM_NAME=iOS. Desktop is built from Code/.")
endif()
# Changes apply to a build-directory copy, retaining the pinned upstream source.
set(JELY_RAYLIB "${CMAKE_CURRENT_BINARY_DIR}/raylib-mobile")
file(COPY "${JELY_CODE}/vendor/raylib-5.5/src/" DESTINATION "${JELY_RAYLIB}")
# ES 3 provides core draw-buffer functions; the ES 2 extension symbol is not
# exported by Android's GLESv3 library or Apple's OpenGLES framework.
file(READ "${JELY_RAYLIB}/rlgl.h" JELY_RLGL)
string(REPLACE "glDrawBuffersEXT(" "glDrawBuffers(" JELY_RLGL "${JELY_RLGL}")
foreach(JELY_ATTACHMENT RANGE 0 7)
  string(REPLACE "GL_COLOR_ATTACHMENT${JELY_ATTACHMENT}_EXT" "GL_COLOR_ATTACHMENT${JELY_ATTACHMENT}" JELY_RLGL "${JELY_RLGL}")
endforeach()
file(WRITE "${JELY_RAYLIB}/rlgl.h" "${JELY_RLGL}")
set(JELY_RAYLIB_SOURCES rcore.c rmodels.c rshapes.c rtext.c rtextures.c utils.c)
list(TRANSFORM JELY_RAYLIB_SOURCES PREPEND "${JELY_RAYLIB}/")
file(READ "${JELY_RAYLIB}/config.h" JELY_CONFIG)
string(REGEX MATCHALL "(^|\n)#define SUPPORT_[A-Z0-9_]+" JELY_SUPPORT "${JELY_CONFIG}")
list(TRANSFORM JELY_SUPPORT REPLACE "\n?#define " "")
list(FILTER JELY_SUPPORT EXCLUDE REGEX "SUPPORT_MODULE_RAUDIO|SUPPORT_GIF_RECORDING|SUPPORT_SCREEN_CAPTURE")
add_library(jely_mobile_raylib STATIC ${JELY_RAYLIB_SOURCES})
target_include_directories(jely_mobile_raylib SYSTEM PUBLIC "${JELY_RAYLIB}")
target_compile_definitions(jely_mobile_raylib PRIVATE EXTERNAL_CONFIG_FLAGS ${JELY_SUPPORT} MAX_MESH_VERTEX_BUFFERS=9 GRAPHICS_API_OPENGL_ES3)
if(ANDROID)
  # Android's raylib stdio wrapper prepends the app directory to relative paths.
  # Already absolute private paths must not have that directory prepended twice.
  file(READ "${JELY_RAYLIB}/utils.c" JELY_ANDROID_UTILS)
  string(REPLACE "fopen(TextFormat(\"%s/%s\", internalDataPath, fileName), mode)" "fopen((fileName[0] == '/')? fileName : TextFormat(\"%s/%s\", internalDataPath, fileName), mode)" JELY_ANDROID_UTILS "${JELY_ANDROID_UTILS}")
  file(WRITE "${JELY_RAYLIB}/utils.c" "${JELY_ANDROID_UTILS}")
  target_compile_definitions(jely_mobile_raylib PRIVATE PLATFORM_ANDROID)
  target_sources(jely_mobile_raylib PRIVATE "${ANDROID_NDK}/sources/android/native_app_glue/android_native_app_glue.c")
  target_include_directories(jely_mobile_raylib PRIVATE "${ANDROID_NDK}/sources/android/native_app_glue")
  target_link_libraries(jely_mobile_raylib PUBLIC android log EGL GLESv3 m)
  file(READ "${JELY_RAYLIB}/platforms/rcore_android.c" JELY_ANDROID_CORE)
  string(REPLACE "EGL_CONTEXT_CLIENT_VERSION, 2," "EGL_CONTEXT_CLIENT_VERSION, 3," JELY_ANDROID_CORE "${JELY_ANDROID_CORE}")
  string(REPLACE "SetupFramebuffer(CORE.Window.display.width, CORE.Window.display.height);" "CORE.Window.screen.width = CORE.Window.display.width;\n    CORE.Window.screen.height = CORE.Window.display.height;\n    SetupFramebuffer(CORE.Window.display.width, CORE.Window.display.height);" JELY_ANDROID_CORE "${JELY_ANDROID_CORE}")
  file(WRITE "${JELY_RAYLIB}/platforms/rcore_android.c" "${JELY_ANDROID_CORE}")
else()
  include(FetchContent)
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  set(SDL_TEST OFF CACHE BOOL "" FORCE)
  set(SDL_TESTS OFF CACHE BOOL "" FORCE)
  set(SDL_JOYSTICK_HIDAPI OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(sdl2 GIT_REPOSITORY https://github.com/libsdl-org/SDL.git GIT_TAG 5d249570393f7a37e037abf22cd6012a4cc56a71)
  FetchContent_MakeAvailable(sdl2)
  target_compile_definitions(jely_mobile_raylib PRIVATE PLATFORM_DESKTOP_SDL)
  target_link_libraries(jely_mobile_raylib PUBLIC SDL2::SDL2-static "-framework OpenGLES")
  # EAGL's onscreen framebuffer is not zero. Route raylib restores and blits to
  # the captured drawable, while preserving ordinary offscreen framebuffer IDs.
  file(READ "${JELY_RAYLIB}/rlgl.h" JELY_RLGL)
  string(REPLACE "#include <GLES3/gl3.h>" "#include <OpenGLES/ES3/gl.h>" JELY_RLGL "${JELY_RLGL}")
  string(REPLACE "#include <GLES2/gl2ext.h>" "#include <OpenGLES/ES3/glext.h>" JELY_RLGL "${JELY_RLGL}")
  # Apple exposes these ES 3 formats through their core names, not ES 2 aliases.
  string(REPLACE "GL_DEPTH_COMPONENT24_OES" "GL_DEPTH_COMPONENT24" JELY_RLGL "${JELY_RLGL}")
  string(REPLACE "GL_DEPTH_COMPONENT32_OES" "GL_DEPTH_COMPONENT32F" JELY_RLGL "${JELY_RLGL}")
  string(REPLACE "GL_DEPTH_COMPONENT, GL_UNSIGNED_INT, NULL)" "GL_DEPTH_COMPONENT, (glInternalFormat == GL_DEPTH_COMPONENT32F)? GL_FLOAT : GL_UNSIGNED_INT, NULL)" JELY_RLGL "${JELY_RLGL}")
  foreach(JELY_CORE_FORMAT R32F RGB32F RGBA32F RED R16F RGB16F RGBA16F)
    string(REPLACE "GL_${JELY_CORE_FORMAT}_EXT" "GL_${JELY_CORE_FORMAT}" JELY_RLGL "${JELY_RLGL}")
  endforeach()
  string(REPLACE "glBindFramebuffer(" "JelyBindFramebuffer(" JELY_RLGL "${JELY_RLGL}")
  string(REPLACE "void rlEnableFramebuffer(unsigned int id)\n{" "extern unsigned int JelyDefaultFramebuffer(void);\nstatic void JelyBindFramebuffer(unsigned int target,unsigned int framebuffer) { glBindFramebuffer(target,framebuffer?framebuffer:JelyDefaultFramebuffer()); }\nvoid rlEnableFramebuffer(unsigned int id)\n{" JELY_RLGL "${JELY_RLGL}")
  file(WRITE "${JELY_RAYLIB}/rlgl.h" "${JELY_RLGL}")
  file(READ "${JELY_RAYLIB}/platforms/rcore_desktop_sdl.c" JELY_SDL_CORE)
  string(REPLACE "SDL_GL_SwapWindow(platform.window);" "extern void JelyPreparePresent(void);\n    JelyPreparePresent();\n    SDL_GL_SwapWindow(platform.window);" JELY_SDL_CORE "${JELY_SDL_CORE}")
  string(REPLACE "TRACELOG(LOG_WARNING, \"GetWindowScaleDPI() not implemented on target platform\");" "/* Mobile drawable uses the SDL window's point dimensions. */" JELY_SDL_CORE "${JELY_SDL_CORE}")
  string(REPLACE "CORE.Window.render.width = CORE.Window.screen.width;" "SDL_GetWindowSize(platform.window, &CORE.Window.screen.width, &CORE.Window.screen.height);\n        CORE.Window.render.width = CORE.Window.screen.width;" JELY_SDL_CORE "${JELY_SDL_CORE}")
  string(REPLACE "CORE.Input.Touch.pointCount = SDL_GetNumTouchFingers(event.touchId);" "CORE.Input.Touch.pointCount = SDL_GetNumTouchFingers(event.touchId);\n    if (CORE.Input.Touch.pointCount > MAX_TOUCH_POINTS) CORE.Input.Touch.pointCount = MAX_TOUCH_POINTS;" JELY_SDL_CORE "${JELY_SDL_CORE}")
  file(WRITE "${JELY_RAYLIB}/platforms/rcore_desktop_sdl.c" "${JELY_SDL_CORE}")
endif()
add_library(jely_mobile_physics STATIC "${JELY_CODE}/src/physics/SoftBody.cpp" "${JELY_CODE}/src/physics/PhysicsWorld.cpp" "${JELY_CODE}/src/scene/Scene.cpp" "${JELY_CODE}/src/scene/Appearance.cpp" "${JELY_CODE}/src/render/Surface.cpp")
target_include_directories(jely_mobile_physics PUBLIC "${JELY_CODE}/include")
file(READ "${JELY_CODE}/assets/InterVariable.ttf" JELY_FONT_HEX HEX)
file(READ "${JELY_CODE}/assets/Inter-OFL.txt" JELY_FONT_LICENSE)
file(READ "${JELY_CODE}/LICENSE" JELY_PROJECT_LICENSE)
string(REGEX REPLACE "(..)" "0x\\1," JELY_FONT_BYTES "${JELY_FONT_HEX}")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
configure_file("${JELY_CODE}/cmake/EmbeddedFont.hpp.in" generated/EmbeddedFont.hpp @ONLY)
set(JELY_APP_SOURCES MobileApp.cpp MobilePlatform.cpp "${JELY_CODE}/src/input/CameraController.cpp" "${JELY_CODE}/src/render/Renderer.cpp" "${JELY_CODE}/src/render/SceneTargets.cpp" "${JELY_CODE}/src/ui/Glass.cpp")
if(ANDROID)
  add_library(3D_Jelly_Physics_Mobile SHARED ${JELY_APP_SOURCES})
  target_include_directories(3D_Jelly_Physics_Mobile PRIVATE "${ANDROID_NDK}/sources/android/native_app_glue")
  target_link_options(3D_Jelly_Physics_Mobile PRIVATE -u ANativeActivity_onCreate -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
else()
  add_executable(3D_Jelly_Physics_Mobile MACOSX_BUNDLE ${JELY_APP_SOURCES} LaunchScreen.storyboard)
  target_compile_definitions(3D_Jelly_Physics_Mobile PRIVATE JELY_IPADOS)
  target_link_libraries(3D_Jelly_Physics_Mobile PRIVATE SDL2::SDL2main)
  set_source_files_properties(LaunchScreen.storyboard PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  # Distinct resource names preserve the notices of every bundled dependency.
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/licenses")
  configure_file("${JELY_CODE}/LICENSE" "${CMAKE_CURRENT_BINARY_DIR}/licenses/MIT.txt" COPYONLY)
  configure_file("${JELY_CODE}/assets/Inter-OFL.txt" "${CMAKE_CURRENT_BINARY_DIR}/licenses/Inter-OFL.txt" COPYONLY)
  configure_file("${JELY_CODE}/vendor/raylib-5.5/LICENSE" "${CMAKE_CURRENT_BINARY_DIR}/licenses/raylib.txt" COPYONLY)
  configure_file("${sdl2_SOURCE_DIR}/LICENSE.txt" "${CMAKE_CURRENT_BINARY_DIR}/licenses/SDL.txt" COPYONLY)
  file(GLOB JELY_BUNDLE_LICENSES "${CMAKE_CURRENT_BINARY_DIR}/licenses/*.txt")
  target_sources(3D_Jelly_Physics_Mobile PRIVATE ${JELY_BUNDLE_LICENSES})
  set_source_files_properties(${JELY_BUNDLE_LICENSES} PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/licenses")
  set_target_properties(3D_Jelly_Physics_Mobile PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in" XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "2" XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER "org.jely.mobile" XCODE_ATTRIBUTE_IPHONEOS_DEPLOYMENT_TARGET "16.0" XCODE_ATTRIBUTE_ASSETCATALOG_COMPILER_APPICON_NAME "AppIcon" XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED "NO")
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/make_icons.py" "${CMAKE_CURRENT_BINARY_DIR}/icons" COMMAND_ERROR_IS_FATAL ANY)
  target_sources(3D_Jelly_Physics_Mobile PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/icons/Assets.xcassets")
  set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/icons/Assets.xcassets" PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
endif()
target_compile_definitions(3D_Jelly_Physics_Mobile PRIVATE JELY_GLES)
target_include_directories(3D_Jelly_Physics_Mobile PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(3D_Jelly_Physics_Mobile PRIVATE jely_mobile_physics jely_mobile_raylib)
target_compile_options(3D_Jelly_Physics_Mobile PRIVATE -Wall -Wextra -Wpedantic)
