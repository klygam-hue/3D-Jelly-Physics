cmake_minimum_required(VERSION 3.25)
project(3D_Jelly_Physics VERSION 1.9.2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(JELY_BUILD_APP "Build the interactive graphics application" ON)
option(JELY_BUILD_TESTS "Build headless physics regression tests" ON)
option(JELY_GRAPHICS_TESTS "Register display-dependent graphics smoke tests" OFF)
set(JELY_VULKAN_DEFAULT OFF)
if((WIN32 AND CMAKE_SIZEOF_VOID_P EQUAL 8 AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64)$") OR (CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|aarch64|arm64)$"))
  set(JELY_VULKAN_DEFAULT ON)
endif()
option(JELY_ENABLE_VULKAN "Include the platform's real ANGLE/Vulkan runtime" ${JELY_VULKAN_DEFAULT})
if(JELY_BUILD_APP AND JELY_ENABLE_VULKAN AND NOT JELY_VULKAN_DEFAULT)
  message(FATAL_ERROR "Pinned ANGLE/Vulkan supports Windows x64 and Linux x64/arm64. macOS currently uses native OpenGL; Metal must not be labeled Vulkan. Configure with -DJELY_ENABLE_VULKAN=OFF.")
endif()
add_library(jely_physics STATIC src/physics/SoftBody.cpp src/physics/PhysicsWorld.cpp src/scene/Scene.cpp src/scene/Appearance.cpp src/render/Surface.cpp)
target_include_directories(jely_physics PUBLIC include)
if(MSVC)
  target_compile_options(jely_physics PRIVATE /W4 /permissive-)
else()
  target_compile_options(jely_physics PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(JELY_BUILD_APP)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(CUSTOMIZE_BUILD ON CACHE BOOL "" FORCE)
  set(USE_AUDIO OFF CACHE BOOL "" FORCE)
  set(SUPPORT_MODULE_RAUDIO OFF CACHE BOOL "" FORCE)
  set(SUPPORT_CLIPBOARD_IMAGE OFF CACHE BOOL "" FORCE)
  set(USE_EXTERNAL_GLFW OFF CACHE STRING "" FORCE)
  set(OPENGL_VERSION 3.3 CACHE STRING "" FORCE)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    # Avoid mandatory Wayland build tools. The package supports X11/XWayland.
    set(GLFW_BUILD_X11 ON CACHE BOOL "" FORCE)
    set(GLFW_BUILD_WAYLAND OFF CACHE BOOL "" FORCE)
  endif()
  add_subdirectory(vendor/raylib-5.5 EXCLUDE_FROM_ALL)
  # raylib 5.5's config parser collects both sides of its skinning conditional.
  # Use its skinning-enabled allocation size once, without altering vendored code.
  get_target_property(JELY_RAYLIB_DEFINITIONS raylib COMPILE_DEFINITIONS)
  list(FILTER JELY_RAYLIB_DEFINITIONS EXCLUDE REGEX "^MAX_MESH_VERTEX_BUFFERS=")
  list(APPEND JELY_RAYLIB_DEFINITIONS MAX_MESH_VERTEX_BUFFERS=9)
  set_target_properties(raylib PROPERTIES COMPILE_DEFINITIONS "${JELY_RAYLIB_DEFINITIONS}")
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/assets/InterVariable.ttf" JELY_FONT_HEX HEX)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/assets/Inter-OFL.txt" JELY_FONT_LICENSE)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE" JELY_PROJECT_LICENSE)
  string(REGEX REPLACE "(..)" "0x\\1," JELY_FONT_BYTES "${JELY_FONT_HEX}")
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
  configure_file(cmake/EmbeddedFont.hpp.in generated/EmbeddedFont.hpp @ONLY)
  add_executable(3D_Jelly_Physics src/main.cpp src/app/Application.cpp src/render/Renderer.cpp src/render/SceneTargets.cpp src/render/Backend.cpp src/render/GraphicsSession.cpp src/input/CameraController.cpp src/ui/Panel.cpp src/ui/Glass.cpp)
  target_include_directories(3D_Jelly_Physics PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/raylib-5.5/src/external/glfw/include")
  target_compile_definitions(raylib PRIVATE JELY_DUAL_GRAPHICS)
  target_compile_definitions(glfw PRIVATE JELY_DUAL_GRAPHICS)
  if(JELY_ENABLE_VULKAN)
    target_compile_definitions(3D_Jelly_Physics PRIVATE JELY_HAS_ANGLE)
  endif()
  if(WIN32 AND JELY_ENABLE_VULKAN)
    enable_language(RC)
    if(MINGW)
      # windres' default preprocessor path handling breaks when GCC lives under a spaced path.
      set(CMAKE_RC_FLAGS "--preprocessor=gcc --preprocessor-arg=-E --preprocessor-arg=-xc --preprocessor-arg=-DRC_INVOKED --use-temp-file")
    endif()
    set(JELY_ANGLE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/vendor/angle")
    set(JELY_ANGLE_FILES libEGL.dll libGLESv2.dll vulkan-1.dll LICENSES.chromium.html)
    set(JELY_ANGLE_HASHES AE32B2441B4AB85CD100FDEAA9DCFD99B1DA15E8FD41C8DE78512C69CB96F596 B4F6D2158D1BD27DED6C65884EFE26005C45E6D15373FF1230A5DF2B6D2DFE00 2D9509B6CAF660F6057BBEADADB8474E740772AB861C8489BADFAD18E76A4056 C1BC6CFDD6C5844720E5E6332698A6131F5402E77ABF1FDB9646740D38065A65)
    set(JELY_ANGLE_DEPENDS)
    foreach(JELY_ASSET_INDEX RANGE 0 3)
      list(GET JELY_ANGLE_FILES ${JELY_ASSET_INDEX} JELY_ASSET_NAME)
      list(GET JELY_ANGLE_HASHES ${JELY_ASSET_INDEX} JELY_ASSET_HASH)
      set(JELY_ASSET_PATH "${JELY_ANGLE_ROOT}/${JELY_ASSET_NAME}")
      if(NOT EXISTS "${JELY_ASSET_PATH}")
        message(FATAL_ERROR "Pinned ANGLE asset missing: ${JELY_ASSET_NAME}. Run bootstrap_angle.ps1.")
      endif()
      file(SHA256 "${JELY_ASSET_PATH}" JELY_ACTUAL_HASH)
      string(TOUPPER "${JELY_ACTUAL_HASH}" JELY_ACTUAL_HASH)
      if(NOT JELY_ACTUAL_HASH STREQUAL JELY_ASSET_HASH)
        message(FATAL_ERROR "Pinned ANGLE integrity mismatch: ${JELY_ASSET_NAME}")
      endif()
      list(APPEND JELY_ANGLE_DEPENDS "${JELY_ASSET_PATH}")
    endforeach()
    configure_file(cmake/GraphicsRuntime.rc.in generated/GraphicsRuntime.rc @ONLY)
    set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/generated/GraphicsRuntime.rc" PROPERTIES OBJECT_DEPENDS "${JELY_ANGLE_DEPENDS}")
    target_sources(3D_Jelly_Physics PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated/GraphicsRuntime.rc")
  endif()
  if(WIN32)
    target_link_libraries(3D_Jelly_Physics PRIVATE bcrypt ole32 uuid)
  elseif(UNIX)
    find_package(Threads REQUIRED)
    target_link_libraries(3D_Jelly_Physics PRIVATE ${CMAKE_DL_LIBS} Threads::Threads)
    if(JELY_ENABLE_VULKAN)
      include(cmake/AngleLinux.cmake)
    endif()
  endif()
  target_include_directories(3D_Jelly_Physics PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_include_directories(3D_Jelly_Physics SYSTEM PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/raylib-5.5/src")
  target_link_libraries(3D_Jelly_Physics PRIVATE jely_physics raylib)
  if(MSVC)
    target_compile_options(3D_Jelly_Physics PRIVATE /W4 /permissive-)
    # Keep the GUI subsystem while using the portable main(argc, argv) entry point.
    target_link_options(3D_Jelly_Physics PRIVATE /ENTRY:mainCRTStartup)
  else()
    target_compile_options(3D_Jelly_Physics PRIVATE -Wall -Wextra -Wpedantic)
    if(WIN32)
      target_link_options(3D_Jelly_Physics PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
  endif()
  if(WIN32)
    set_target_properties(3D_Jelly_Physics PROPERTIES WIN32_EXECUTABLE TRUE)
  endif()
  set_target_properties(3D_Jelly_Physics PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>")
  if(APPLE)
    set_target_properties(3D_Jelly_Physics PROPERTIES MACOSX_BUNDLE TRUE MACOSX_BUNDLE_GUI_IDENTIFIER "org.3djely.lab" MACOSX_BUNDLE_BUNDLE_NAME "3D Jelly Physics" MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}" MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}" MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/cmake/Info.plist.in")
    install(TARGETS 3D_Jelly_Physics BUNDLE DESTINATION .)
  else()
    install(TARGETS 3D_Jelly_Physics RUNTIME DESTINATION Release)
  endif()
  install(FILES assets/Inter-OFL.txt vendor/raylib-5.5/LICENSE README.md DESTINATION licenses)
  install(FILES LICENSE DESTINATION licenses RENAME 3D-Jelly-Physics-MIT.txt)
endif()
if(JELY_BUILD_TESTS)
  enable_testing()
  add_executable(jely_tests tests/PhysicsTests.cpp)
  target_link_libraries(jely_tests PRIVATE jely_physics)
  if(MINGW)
    target_link_options(jely_tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME physics_regression COMMAND jely_tests)
  if(JELY_BUILD_APP AND JELY_GRAPHICS_TESTS)
    add_test(NAME opengl_scene_smoke COMMAND 3D_Jelly_Physics --backend opengl --strict-backend --smoke 300 --automate --report "${CMAKE_CURRENT_BINARY_DIR}/opengl-scene.txt")
    add_test(NAME opengl_ui_smoke COMMAND 3D_Jelly_Physics --backend opengl --strict-backend --smoke 480 --ui-automate --report "${CMAKE_CURRENT_BINARY_DIR}/opengl-ui.txt")
    add_test(NAME opengl_spawn_smoke COMMAND 3D_Jelly_Physics --backend opengl --strict-backend --smoke 300 --spawn-automate --spawn-seed 431 --report "${CMAKE_CURRENT_BINARY_DIR}/opengl-spawn.txt")
    add_test(NAME opengl_ground_smoke COMMAND 3D_Jelly_Physics --backend opengl --strict-backend --smoke 350 --ground-automate --report "${CMAKE_CURRENT_BINARY_DIR}/opengl-ground.txt")
    add_test(NAME opengl_softness_smoke COMMAND 3D_Jelly_Physics --backend opengl --strict-backend --smoke 350 --softness-automate --report "${CMAKE_CURRENT_BINARY_DIR}/opengl-softness.txt")
    set_tests_properties(opengl_softness_smoke PROPERTIES LABELS graphics TIMEOUT 90 RUN_SERIAL TRUE)
    set_tests_properties(opengl_ground_smoke PROPERTIES LABELS graphics TIMEOUT 90 RUN_SERIAL TRUE)
    set_tests_properties(opengl_spawn_smoke PROPERTIES LABELS graphics TIMEOUT 90 RUN_SERIAL TRUE)
    set_tests_properties(opengl_scene_smoke opengl_ui_smoke PROPERTIES LABELS graphics TIMEOUT 90 RUN_SERIAL TRUE)
  endif()
endif()
